# Hebrew i18n Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the Maya AI dashboard fully Hebrew/RTL by default, with a header toggle to switch to English — without touching any API calls, data logic, or internal types.

**Architecture:** A single `i18n.ts` file holds all translated strings. A `LanguageContext` provides `useLanguage()` to all client components. Server page components are left unchanged for data fetching; they pass data to new or existing client components that consume `useLanguage()`. A `DashboardShell` client component wraps the layout to apply `dir="rtl"/"ltr"` and provide the context.

**Tech Stack:** Next.js App Router, React Context, TypeScript, Tailwind CSS

// dashboard/context/language-context.tsx
"use client";

import { createContext, useContext, useState } from "react";
import { translations, type Lang, type Translations } from "@/lib/i18n";

interface LanguageContextValue {
  lang: Lang;
  setLang: (l: Lang) => void;
  t: Translations;
}

const LanguageContext = createContext<LanguageContextValue>({
  lang: "he",
  setLang: () => {},
  t: translations.he,
});

export function LanguageProvider({ children }: { children: React.ReactNode }) {
  const [lang, setLang] = useState<Lang>("he");
  return (
    <LanguageContext.Provider value={{ lang, setLang, t: translations[lang] }}>
      {children}
    </LanguageContext.Provider>
  );
}

export function useLanguage() {
  return useContext(LanguageContext);
}

export function TestAgent({ agents }: Props) {
  const [selectedAgent, setSelectedAgent] = useState(agents[0]?.id ?? "");

    setMessages([{ role: "assistant", content: agent.first_message }]);

        method: "POST",
        headers: { "Content-Type": "application/json" },

      });

        setMessages((prev) => [...prev, { role: "assistant", content: reply }]);
      } else {
        setMessages((prev) => [...prev, { role: "assistant", content: t.backend_error }]);
